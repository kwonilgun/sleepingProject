// src/screens/PlayerScreen.tsx
import React, { useState, useEffect, useMemo } from 'react';
import { View, StyleSheet, TouchableOpacity, Text, Alert, Platform, PermissionsAndroid } from 'react-native';
import FontAwesome from 'react-native-vector-icons/FontAwesome';
import colors from '../../styles/colors';
import { RFPercentage } from 'react-native-responsive-fontsize';
import HeaderComponent from '../../utils/basicForm/HeaderComponents';
import WrapperContainer from '../../utils/basicForm/WrapperContainer';
import MaterialIcon from 'react-native-vector-icons/MaterialCommunityIcons';

import TrackPlayer, {
  State,
} from 'react-native-track-player';
import { PlayerScreenProps } from '../model/types/TUserNavigator';

// Custom Hooks
import { useTrackPlayerSetup } from '../../hooks/useTrackPlayerSetup';
import { useSleepTimer } from '../../hooks/useSleepTimer';

// Components
import PlaybackControls from './PlaybackControls';
import TrackProgressSlider from './TrackProgressSlider';
import VolumeControl from './VolumeControl';

// Enums and Interfaces (can be moved to a separate types file if they grow)
export interface PlaylistItem {
  id: string;
  name: string;
  title?: string;
  artist?: string;
  path: string;
  type: 'file' | 'folder';
  uri?: string;
  duration?: number;
  isSelected?: boolean;
  isDirectoryOpen?: boolean;
  children?: PlaylistItem[];
  depth?: number;
}

enum RepeatMode {
  Off,
  RepeatOne,
  RepeatAll,
}

const PlayerScreen: React.FC<PlayerScreenProps> = ({ route, navigation }) => {
  const { selectedTracks, playlist } = route.params;

  // console.log('PlayerScreen selectedTracks: ', selectedTracks);
  // console.log('playlist : ', playlist);

  const [displayTitle, setDisplayTitle] = useState('선택된 곡 없음');
  const [currentTrackIndex, setCurrentTrackIndex] = useState(0);

  // Use custom hook for TrackPlayer setup and playback state
  const {
    playbackState,
    progress,
    repeatMode,
    volume,
    isMuted,
    togglePlayback,
    handleSkipPrevious,
    handleSkipNext,
    toggleRepeatMode,
    setVolume,
    toggleMute,
    isPlaying,
    isLoading,
  } = useTrackPlayerSetup(selectedTracks, playlist, currentTrackIndex, setCurrentTrackIndex);

  // Use custom hook for sleep timer functionality
  const { sleepTimerActive, startSleepTimer } = useSleepTimer(
    playbackState,
    isPlaying,
    togglePlayback,
    navigation
  );


  // Update display title when currentTrackIndex or playlist changes
  useEffect(() => {
    if (selectedTracks.length > 0 && currentTrackIndex < selectedTracks.length) {
      const currentUri = selectedTracks[currentTrackIndex];
      const currentTrack = playlist.find(item => item.uri === currentUri && item.type === 'file');
      if (currentTrack) {
        console.log('currentTrack :', currentTrack);
        setDisplayTitle(currentTrack.name.replace(/\.mp3$/i, ''));
      }
    } else {
      setDisplayTitle('선택된 곡 없음');
    }
  }, [currentTrackIndex, selectedTracks, playlist]);


  const LeftCustomComponent = () => {
    const handleGoBack = () => {
      // Navigation back logic (consider if you need to stop player here or in useTrackPlayerSetup cleanup)
      console.log('handleGoBack...');
      navigation.navigate('PlaylistScreen');
    };

    return (
      <TouchableOpacity onPress={handleGoBack}>
        <FontAwesome
          style={styles.backButtonIcon}
          name="arrow-left"
        />
      </TouchableOpacity>
    );
  };

  return (
    <WrapperContainer containerStyle={styles.wrapperContainer}>
      <HeaderComponent
        rightPressActive={false}
        isLeftView={true}
        leftCustomView={LeftCustomComponent}
        centerText="🎵 음악 "
        containerStyle={styles.headerContainer}
        isRight={false}
      />
      <View style={styles.container}>
        <Text style={styles.title}>🎧 제목:</Text>
        <Text style={styles.nowPlaying} numberOfLines={1} ellipsizeMode="tail">
          {displayTitle}
          {isLoading && ' (로딩 중...)'}
        </Text>

        <PlaybackControls
          isPlaying={isPlaying}
          togglePlayback={togglePlayback}
          handleSkipPrevious={handleSkipPrevious}
          handleSkipNext={handleSkipNext}
          toggleRepeatMode={toggleRepeatMode}
          repeatMode={repeatMode}
          disabled={!selectedTracks || selectedTracks.length === 0}
        />

        {selectedTracks.length > 0 && (
          <TrackProgressSlider
            progress={progress}
            onSlidingComplete={(value) => TrackPlayer.seekTo(value)}
            disabled={!selectedTracks || selectedTracks.length === 0 || isLoading}
          />
        )}

        <VolumeControl
          volume={volume}
          isMuted={isMuted}
          setVolume={setVolume}
          toggleMute={toggleMute}
        />

        <TouchableOpacity
          onPress={startSleepTimer}
          style={[
            styles.sleepModeButton,
            sleepTimerActive ? styles.sleepModeButtonActive : styles.sleepModeButtonInactive,
          ]}
        >
          <MaterialIcon
            name="sleep"
            size={RFPercentage(3)}
            color="white"
          />
          <Text style={styles.sleepModeButtonText}>
            {sleepTimerActive ? '수면모드 활성화됨' : '수면모드'}
          </Text>
        </TouchableOpacity>

      </View>
    </WrapperContainer>
  );
};

const styles = StyleSheet.create({
  wrapperContainer: {
    paddingHorizontal: 0,
  },
  headerContainer: {
    paddingHorizontal: 8,
  },
  backButtonIcon: {
    height: RFPercentage(8),
    width: RFPercentage(10),
    marginTop: RFPercentage(2),
    color: colors.black,
    fontSize: RFPercentage(5),
    fontWeight: 'bold',
  },
  container: {
    flex: 1,
    padding: 20,
    justifyContent: 'center',
    backgroundColor: colors.lightGrey,
  },
  title: {
    fontSize: RFPercentage(2.5),
    fontWeight: 'bold',
    marginBottom: 20,
    textAlign: 'center',
    color: '#2c3e50',
  },
  nowPlaying: {
    fontSize: RFPercentage(2.5),
    marginBottom: 15,
    textAlign: 'center',
    fontStyle: 'italic',
    color: '#34495e',
  },
  sleepModeButton: {
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'center',
    marginTop: 30,
    paddingVertical: 12,
    paddingHorizontal: 20,
    borderRadius: 10,
    alignSelf: 'center',
  },
  sleepModeButtonInactive: {
    backgroundColor: '#6a0dad',
  },
  sleepModeButtonActive: {
    backgroundColor: '#c0a0ff',
  },
  sleepModeButtonText: {
    color: 'white',
    fontSize: RFPercentage(2.2),
    fontWeight: 'bold',
    marginLeft: 10,
  },
});

export default PlayerScreen;
